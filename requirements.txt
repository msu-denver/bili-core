# =============================================================================
# bili-core — Lean Core Dependencies
# =============================================================================
# This file declares the BASE install requirements: everything that
# ``import bili.iris`` and ``import bili.aether`` need to run, and nothing
# more.  Heavy optional backends (databases, ML frameworks, vector search,
# UI surfaces, auth providers) live ONLY in the extras_require section of
# setup.py and must NOT appear here.
#
# Extras catalogue (see setup.py for full pin list):
#   pip install bili-core[streamlit]    # Streamlit UI surface
#   pip install bili-core[flask]        # Flask REST API surface
#   pip install bili-core[aegis]        # AEGIS adversarial testing
#   pip install bili-core[faiss]        # FAISS vector search
#   pip install bili-core[opensearch]   # Amazon OpenSearch
#   pip install bili-core[mongo]        # MongoDB checkpointer
#   pip install bili-core[postgres]     # PostgreSQL checkpointer
#   pip install bili-core[huggingface]  # HuggingFace local models
#   pip install bili-core[llamacpp]     # llama.cpp local models
#   pip install bili-core[ml]           # Full ML stack (AEGIS attacks)
#   pip install bili-core[docs]         # Document-processing tools
#   pip install bili-core[firebase]     # Firebase auth provider
#   pip install bili-core[mcp]          # MCP client subsystem
#   pip install bili-core[all-providers]# All API LLM providers
#   pip install bili-core[all]          # Everything (backward-compat bundle)
#
# Dev tooling is declared in the [dev] extra (setup.py); install it with:
#   pip install bili-core[dev]

# -----------------------------------------------------------------------------
# LangChain & LangGraph Core
# -----------------------------------------------------------------------------
langchain~=1.5.2
langchain-core~=1.8.5
langchain-community~=0.4.2
langgraph~=1.2.24

# -----------------------------------------------------------------------------
# Primary LLM Provider SDKs
# These are the built-in providers IRIS ships support for out of the box.
# Additional providers (Anthropic, Mistral, Cohere, etc.) are in extras.
# -----------------------------------------------------------------------------
langchain-aws~=1.10.0
# 3.2.5 is the first release without the pyarrow dependency; earlier lines cap
# pyarrow below the first release that ships CPython 3.14 wheels (22.0.0).
langchain-google-vertexai~=3.3.0
langchain-openai~=1.8.5

# -----------------------------------------------------------------------------
# Core Utilities
# -----------------------------------------------------------------------------
# jsonschema powers post-hoc validation in
# bili.iris.providers.structured_output.parse_structured_content.
jsonschema~=4.21
pyyaml~=6.0
# Floor set by google-api-core (2.35 and later require requests>=2.33). A
# patch-level ~=2.32.x pin here makes every newer google-api-core unresolvable.
requests~=2.33

# =============================================================================
# Pinned Transitive Dependencies
# =============================================================================
# Packages below are transitive dependencies pinned to prevent pip
# backtracking during resolution.  They are NOT imported directly by
# bili-core runtime code; they are here solely for install-time stability.
# Do NOT add new entries here without confirming they are genuine transitives
# of the lean-core packages above.

# LangChain ecosystem
# langgraph-prebuilt releases on its own cadence and has imported symbols that
# were absent from the langgraph release it was installed next to. Move it only
# together with langgraph: langgraph 1.2.24 requires >=1.1.3,<1.2.
langgraph-prebuilt==1.1.3

# Runtime glue
anyio==4.16.0

# Google Cloud packages (from langchain-google-vertexai)
# These move together as one set. They share protobuf, grpcio and
# google-api-core constraints, so bumping one alone fails to resolve.
# - protobuf is left unpinned. google-api-core, googleapis-common-protos,
#   proto-plus, grpc-google-iam-v1 and google-cloud-resource-manager in this set
#   require protobuf>=6.33.5, and google-cloud-aiplatform 1.x caps it below 7,
#   so it resolves to the 6.x line, which the [streamlit] extra also requires.
# - google-cloud-aiplatform stays on 1.x: langchain-google-vertexai needs only
#   >=1.97, and 2.x would admit protobuf 7 in the lean core while [all] stays
#   on 6.
# - google-genai stays below 3 because google-cloud-aiplatform requires
#   google-genai<3. The [google-genai] extra in setup.py is capped to match.
# - google-crc32c 1.8.0 is the first release with CPython 3.14 wheels.
google-api-core==2.44.0
google-cloud-aiplatform==1.165.1
google-cloud-resource-manager==1.19.0
google-cloud-storage==3.17.2
google-crc32c==1.9.0
google-genai==2.34.0
google-resumable-media==2.11.0
googleapis-common-protos==1.76.0
grpc-google-iam-v1==0.14.6
proto-plus==1.29.0
